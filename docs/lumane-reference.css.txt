function rosette(cv,{R=120,r=33,layers=16,ink="31,74,55",alpha=.2,p=1}={}){
  const dpr=Math.min(devicePixelRatio||1,2), S=cv.clientWidth; if(!S) return;
  cv.width=cv.height=Math.round(S*dpr);
  const c=cv.getContext("2d"); c.setTransform(dpr,0,0,dpr,S/2*dpr,S/2*dpr);
  const gcd=(a,b)=>b?gcd(b,a%b):a, g=gcd(R,r), T=2*Math.PI*r/g, k=(R-r)/r, lobes=R/g, sc=(S/2-2)/R, steps=Math.ceil(T*40);
  c.lineWidth=.55;
  for(let i=0;i<layers;i++){
    const f=i/layers, d=r*(.35+.65*f), rot=f*2*Math.PI/lobes, cr=Math.cos(rot), sr=Math.sin(rot);
    c.strokeStyle=`rgba(${ink},${alpha*(.4+.6*Math.sin(f*Math.PI))})`;
    const n=Math.floor(steps*Math.min(1,Math.max(0,p*1.4-f*.4)));
    c.beginPath();
    for(let j=0;j<=n;j++){ const t=j/steps*T, x=(R-r)*Math.cos(t)+d*Math.cos(k*t), y=(R-r)*Math.sin(t)-d*Math.sin(k*t);
      const X=(x*cr-y*sr)*sc, Y=(x*sr+y*cr)*sc; j?c.lineTo(X,Y):c.moveTo(X,Y); }
    c.stroke();
  }
}
